cmake_minimum_required(VERSION 3.24)
project(GinPerformanceBaseline LANGUAGES CXX)
include(CTest)
if(MINGW AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 16 AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 17)
    add_compile_definitions(_GLIBCXX_ATOMIC_WORD_BUILTINS=1)
endif()

set(GIN_CPP_ROOT "" CACHE PATH "gin_cpp source checkout")
if(NOT EXISTS "${GIN_CPP_ROOT}/gin/gin.h")
    message(FATAL_ERROR "Set GIN_CPP_ROOT to a gin_cpp checkout")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GIN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GIN_BUILD_MOUDU_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${GIN_CPP_ROOT}" gin-cpp)
add_executable(disk_http_benchmark main.cpp)
target_link_libraries(disk_http_benchmark PRIVATE gin::gin)
